# zenvibelaptop: Ryzen 7 5825U 8C/16T, 32G RAM — server-focused UKGeo profile
# Heap 8G start; raise to 10G only if GC logs show pressure.
-Xms8G
-Xmx8G

# G1: prefer Java 21 defaults; tune pause target before forcing generation sizes.
-XX:+UseG1GC

-XX:+ParallelRefProcEnabled

-XX:+AlwaysPreTouch
-XX:+PerfDisableSharedMem

-Xlog:gc*:file=logs/gc.log:time,uptime,level,tags:filecount=5,filesize=10M
